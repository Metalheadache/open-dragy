cmake_minimum_required(VERSION 3.20)
project(dragy_esp32_reference VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(REFERENCE_SANITIZERS "Address and undefined-behavior sanitizers" ON)
add_library(dragy_reference src/telemetry.cpp src/drag.cpp src/speed_filter.cpp src/geometry.cpp)
target_include_directories(dragy_reference PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
  target_compile_options(dragy_reference PRIVATE -Wall -Wextra -Werror)
  if(REFERENCE_SANITIZERS)
    target_compile_options(dragy_reference PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(dragy_reference PUBLIC -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
find_package(Threads REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
foreach(name decoder speed_filter rate ble_session launch curved_profiles drag consistency geometry)
  add_executable(${name}_test tests/${name}_test.cpp)
  target_link_libraries(${name}_test PRIVATE dragy_reference Threads::Threads)
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
add_executable(reference_replay tools/replay.cpp)
target_link_libraries(reference_replay PRIVATE dragy_reference)
add_executable(reference_generate tools/generate.cpp)
target_link_libraries(reference_generate PRIVATE dragy_reference)
add_executable(reference_benchmark tools/benchmark.cpp)
target_link_libraries(reference_benchmark PRIVATE dragy_reference)
add_test(NAME replay COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_replay.py" "${CMAKE_CURRENT_BINARY_DIR}")
